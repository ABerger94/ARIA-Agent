# ARIA v1 — dependency pins (Fork A, Step 1: audit AUDIT-FINDINGS §3B-7).
#
# Supported runtime: Python 3.12 on Windows ONLY, always via the py
# launcher — `aria/speech.py` imports audioop (removed in Python 3.13),
# and the laptop's default `python`/`pip` is 3.14, where PyAudio has no
# wheel. Never install with bare `pip` / `python`.
#
#   Install:  py -3.12 -m pip install -r requirements.txt
#
# WARNING: several pins below were carried from the audit and are NOT yet
# verified against the laptop's actual environment. Regenerate this file
# from the working laptop before treating it as final:
#   py -3.12 -m pip freeze
# and commit the result (keeping this header).
#
# Installability audit (2026-10-21, crash round 4): EVERY pin below was
# checked against the PyPI JSON API for a CP312/win_amd64-compatible
# wheel (the laptop has no MSVC/Rust toolchain, so a pin whose only
# artifacts are sdists or other-platform wheels can never install).
# Bumps from that audit are commented at their pins.
opencv-python==4.8.0.76
numpy==1.26.2
# COM ISOLATION (crash round 5, 2026-10-21): pyttsx3 (below),
# pycaw, comtypes and pywin32 are used ONLY by the helper process
# (aria/comhost.py) — the main process imports none of them; a
# static test enforces it. The pins stay because the helper needs
# them on the laptop.
pyttsx3==2.90
SpeechRecognition==3.10.0
# psutil 5.9.0 predates Python 3.12: its newest Windows wheel is
# cp310, so the laptop fell back to a C-extension source build with
# no compiler and failed. 5.9.4 is the minimal bump shipping a
# compatible wheel (psutil-5.9.4-cp36-abi3-win_amd64.whl, stable ABI
# loads on 3.12); every ARIA call site (cpu_percent, virtual_memory,
# disk_usage, disk_partitions, sensors_battery, net_io_counters,
# boot_time, Process) verified unchanged against 5.9.4.
psutil==5.9.4
Pillow==10.0.0
edge-tts==7.3.1
pygame==2.6.1
# UI host (Merger Phase 4e): PyQt6 window hosting the frame canvas —
# the only host that can receive real OS drag-and-drop (cv2.imshow
# windows never deliver drop events). ARIA_UI_HOST=cv2, or a missing /
# failed Qt, falls back to the classic window. Floor pin: the laptop
# already satisfies this (PyQt6 installed for Mark-XXXIX-OR).
PyQt6>=6.6.0
pyserial==3.5
# pyautogui is sdist-only on PyPI (0.9.54 is the latest upstream
# release — no wheel-shipping version exists to bump to). It is
# PURE Python: its sdist builds a py3-none-any wheel with plain
# setuptools, no compiler (verified 2026-10-21), so it does not hit
# the toolchain failure class. Same for pygetwindow below.
pyautogui==0.9.54
pycaw==20230407; sys_platform == "win32"
# comtypes is imported directly by aria/comhost.py (the helper
# process) but used to arrive only transitively via pycaw — pin it
# explicitly: pycaw 20230407 is the known-good pairing for 1.2.0.
# NOTE (2026-10-21, laptop-verified): the Release() crashes happened
# WITH 1.2.0 installed — the fault was cross-thread finalization of
# comtypes wrappers, not the version; round 5 contains it by
# process isolation (aria/comhost.py is the only COM consumer).
comtypes==1.2.0; sys_platform == "win32"
pywin32==306; sys_platform == "win32"
# Mic capture (Fork A, Step 2): sounddevice replaces PyAudio/sr.Microphone.
# ABI-independent wheel with PortAudio bundled — no per-Python wheel trap.
sounddevice==0.5.6
pystray==0.19.0
pygetwindow==0.0.9; sys_platform == "win32"
# duckduckgo-search 6.0.0 depends on pyreqwest-impersonate, which is
# sdist-only -> maturin tries to bootstrap Rust on the laptop and
# dies (the 2026-10-21 install failure). Later releases use primp

# 6.2.4 (first primp release) and 7.5.5 both crash DDGS() against
# primp 2.x ("Invalid impersonate"), and 8.0.0 fails the same way.
# 8.0.1 is the minimal version verified end-to-end: wheel
# duckduckgo_search-8.0.1-py3-none-any.whl; resolves primp 2.0.1
# (cp310-abi3-win_amd64) + lxml 6.1.3 (cp312-win_amd64); DDGS()
# constructs, and ARIA's call — DDGS().text(query, max_results=3),
# consuming r['title']/r['body'] — binds unchanged (result dicts
# still carry title/href/body).
duckduckgo-search==8.0.1
aiohttp==3.9.0
requests==2.31.0
cryptography==41.0.0
# Optional: MCP client support (mcp_* tools). Pin <2: the v2 SDK renamed
